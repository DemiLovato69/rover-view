cmake_minimum_required(VERSION 3.24)
project(FieldRover VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Quick QuickControls2 Network Sql Concurrent Sensors)
if(NOT ANDROID)
  find_package(Qt6 REQUIRED COMPONENTS SerialPort)
endif()
find_package(PROJ CONFIG REQUIRED)
include(FetchContent)
FetchContent_Declare(rtklib URL https://codeload.github.com/tomojitakasu/RTKLIB/tar.gz/180043ee24b6d2b168f98b64be15f69d50046b1a URL_HASH SHA256=e8231a6931b0c5d12fb26368572cd3fc515fbd09d4a1c7597ed4955f15a10113)
FetchContent_MakeAvailable(rtklib)
set(src ${rtklib_SOURCE_DIR}/src)
file(GLOB receivers ${src}/rcv/*.c)
add_library(rtk STATIC ${receivers})
# b34 uses four uninitialized atmospheric corrections in rescode's first iteration.
file(READ ${src}/pntpos.c pntpos)
string(REPLACE "dion,dtrp,vmeas,vion,vtrp" "dion=0,dtrp=0,vmeas,vion=0,vtrp=0" pntpos "${pntpos}")
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/pntpos.c CONTENT "${pntpos}" @ONLY)
target_sources(rtk PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/pntpos.c)
foreach(f rtkcmn rtcm rtcm2 rtcm3 rtcm3e rcvraw ephemeris preceph rtkpos lambda ppp ppp_ar sbas ionex tides solution geoid rinex)
  target_sources(rtk PRIVATE ${src}/${f}.c)
endforeach()
target_include_directories(rtk PUBLIC ${src})
target_compile_definitions(rtk PUBLIC ENAGLO ENAGAL ENACMP)
if(WIN32)
  target_compile_definitions(rtk PUBLIC WIN32 NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(rtk PUBLIC winmm ws2_32)
endif()
set(grid ${CMAKE_CURRENT_BINARY_DIR}/ca_nrc_CGG2013an83.tif)
file(DOWNLOAD https://cdn.proj.org/ca_nrc_CGG2013an83.tif ${grid} EXPECTED_HASH SHA256=a129e9e9b023bb5c7133b3a91e41f1d528892fc28e99dcecb251cd71292968cb TLS_VERIFY ON)
qt_standard_project_setup(REQUIRES 6.5)
set(collector_sources src/solver.cpp src/link.cpp src/survey.cpp src/survey_fieldwork.cpp src/fieldwork.cpp src/surface.cpp src/basemap.cpp src/devicelevel.cpp src/tinitem.cpp)
set(collector_libs Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Sql Qt6::Concurrent Qt6::Sensors PROJ::proj rtk)
qt_add_executable(fieldrover src/main.cpp ${collector_sources})
set_target_properties(fieldrover PROPERTIES MACOSX_BUNDLE TRUE WIN32_EXECUTABLE TRUE RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
qt_add_resources(fieldrover grid PREFIX / BASE ${CMAKE_CURRENT_BINARY_DIR} FILES ${grid})
set(fixtures fixtures/practice.bin fixtures/grading-walk.bin fixtures/practice-pad.landxml fixtures/practice-aerial.mbtiles fixtures/practice-trench.dxf)
qt_add_resources(fieldrover practice PREFIX / FILES ${fixtures} NOTICE.md licenses/testing.txt licenses/RTKLIB.txt licenses/usb-serial.txt licenses/delaunator.txt)
qt_add_qml_module(fieldrover URI FieldRover VERSION 1.0 QML_FILES Main.qml FieldPlot.qml GradeSetup.qml GradeBar.qml LineGauge.qml FieldButton.qml FieldCombo.qml FieldDialog.qml ChoiceStrip.qml ToolGlyph.qml SurfacePreview.qml FieldTools.qml PlanSketch.qml InstrumentDock.qml WorkToolbar.qml LineDraft.qml LineEditor.qml)
target_link_libraries(fieldrover PRIVATE ${collector_libs})
if(ANDROID)
  FetchContent_Declare(usbserial URL https://codeload.github.com/mik3y/usb-serial-for-android/tar.gz/e1018ab31c118b7d2e15fc7c2e9c5b19f33eb1f1 URL_HASH SHA256=5fc1d7738bdd9396de189fcd3690a62629623f893d24ea2cb7da749ea1c88b80)
  FetchContent_MakeAvailable(usbserial)
  set(pkg ${CMAKE_BINARY_DIR}/android-package)
  file(COPY android/ DESTINATION ${pkg})
  file(COPY ${usbserial_SOURCE_DIR}/usbSerialForAndroid/src/main/java/ DESTINATION ${pkg}/src)
  set_target_properties(fieldrover PROPERTIES QT_ANDROID_PACKAGE_SOURCE_DIR ${pkg} QT_ANDROID_PACKAGE_NAME org.fieldrover.collector QT_ANDROID_MIN_SDK_VERSION 28)
else()
  target_link_libraries(fieldrover PRIVATE Qt6::SerialPort)
endif()
include(CTest)
if(BUILD_TESTING AND NOT ANDROID)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(collector_test tests/collector_test.cpp ${collector_sources})
  qt_add_resources(collector_test testgrid PREFIX / BASE ${CMAKE_CURRENT_BINARY_DIR} FILES ${grid})
  qt_add_resources(collector_test fixture PREFIX / FILES ${fixtures})
  target_include_directories(collector_test PRIVATE src)
  target_link_libraries(collector_test PRIVATE Qt6::Test Qt6::SerialPort ${collector_libs})
  add_test(NAME collector COMMAND collector_test WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
  set_tests_properties(collector PROPERTIES TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()
install(TARGETS fieldrover BUNDLE DESTINATION . RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
install(FILES NOTICE.md DESTINATION share/fieldrover)
install(DIRECTORY licenses DESTINATION share/fieldrover)
if(WIN32)
  install(DIRECTORY "$<TARGET_FILE_DIR:fieldrover>/" DESTINATION bin FILES_MATCHING PATTERN "*.dll")
endif()
# macOS is the local development target; its bundle uses the installed Qt and PROJ.
if(NOT ANDROID AND NOT APPLE)
  qt_generate_deploy_qml_app_script(TARGET fieldrover OUTPUT_SCRIPT deploy NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy})
endif()
